def mutant_root(file, old, new):
    d=tempfile.mkdtemp(prefix="csvtotal-mut-")
    shutil.copytree(f"{WT}/src", f"{d}/src")
    shutil.copy(f"{WT}/package.json", d)
    p=f"{d}/src/{file}"; t=open(p).read()

    open(p,"w").write(t.replace(old,new))
    return d
def capture(fname, title, scenario, cases, probe_desc, probe, mutation):
    # cases: list of (label, filename, content or None, args-builder)
    tmp=tempfile.mkdtemp(prefix="csvtotal-qa-")
    lines=[f"# F2 wave 1 {title}","",f"- Change: csv-total (F2 wave 1, Phase 1 QA)",f"- Head: `{HEAD}` (working tree clean at capture: {'yes' if clean else 'NO'})",
           f"- Captured: {datetime.datetime.now().astimezone().isoformat(timespec='seconds')}",f"- Runtime: bun {BUNV}",f"- Scenario: {scenario}",""]
    results=[]
    def one(root, label):
        out=[]
        ok=True
        for (clabel, fn, content, argsf) in cases:
            if content is not None:
                open(f"{tmp}/{fn}","w").write(content)
            args=argsf(f"{tmp}/{fn}" if fn else None)
            rc,so,se=run_cli(args, root)
            shown=" ".join(a.replace(tmp+"/","") for a in args)
            good=probe(clabel, rc, so, se)
            ok=ok and good
            out.append((clabel, fn, content, shown, rc, so, se, good))
        return ok,out
    ok_head,out_head=one(None,"head")
    lines.append("## Cases on head\n")
    for (clabel, fn, content, shown, rc, so, se, good) in out_head:
        lines.append(f"### {clabel}\n")
        lines.append(f"- Command: `bun src/cli.ts {shown}` (cwd = repo root; input file in a temp dir)")
        if content is not None:
            lines += [f"- Input repr: `{content!r}`"]
        lines += [f"- Exit code: `{rc}`", f"- stdout repr: `{so!r}`", f"- stderr repr: `{se!r}`", f"- Case probe: {'MATCH' if good else 'NO MATCH'}",""]
    lines += ["## Probe","",f"- Expected: {probe_desc}",f"- Probe result on head: {'MATCH' if ok_head else 'NO MATCH'}",""]
    file, old, new, why = mutation

--- script ---
const D = ".omp/team-config/evidence/gate/plan-review-loop-measurement/experiment";
const d = await Bun.file(D + "/dataset.json").json();
const s = await Bun.file(D + "/summary.json").json();
const f = r => (r === null ? "n/a" : r.value === null ? `n/a (${r.reason})` : `${r.numerator}/${r.denominator}`);
const intended = { "H-Focus": 5, "H-Replay": 15, S: 15, T: 5 };
for (const a of s.arms) {
	const trials = d.trials.filter(t => t.arm === a.arm);
	const tri = {};
	for (const t of trials) tri[t.triage] = (tri[t.triage] ?? 0) + 1;
	console.log(`arm ${a.arm}: intended ${intended[a.arm]} completed ${trials.filter(t => t.status === "completed").length} trial.triage ${JSON.stringify(tri)}`);
}
for (const a of s.arms) {
	console.log(`${a.arm}: falseDrop ${f(a.falseDrop)}, falsePositive ${f(a.falsePositive)}, triage ${JSON.stringify(a.triage)}, offLedger ${a.additionalBlocking === null ? "n/a (triage unavailable)" : JSON.stringify(a.additionalBlocking)}`);
}
const armOf = new Map(d.trials.map(t => [t.id, t.arm]));
const fOf = new Map(d.findings.map(x => [x.id, x]));
const add = d.adjudications.filter(a => a.kind === "additional-defect" && a.validity === "valid");
const by = {};
for (const a of add) {
	const arm = armOf.get(fOf.get(a.finding).trial);
	by[arm] ??= { valid: 0, blocking: 0, below: 0 };
	by[arm].valid++;
	a.blocking === "blocking" ? by[arm].blocking++ : by[arm].below++;
}
console.log(`valid additional defects by arm: ${JSON.stringify(by)}`);
